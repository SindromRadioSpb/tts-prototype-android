```powershell
node docs/research/engineering-atlas/2026-11-05-a97001/compile-editorial.cjs

node scripts/atlas/build-search-index.cjs data/atlas/a97001
node scripts/atlas/build-train-bank.cjs --data=data/atlas/a97001
```
